> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Give the app real curriculum data: a validated JSON/Markdown content format, build tools that turn it into a small manifest plus lazy-loaded indexes, typed loaders, and a seed bank of 33 lessons and 660 questions for CLF-C02 and AIF-C01.

**Architecture:** Authored content lives in `content/<cert>/`. Node tools in `tools/` validate it with Zod schemas shared from `src/content/schema.js`, then write `src/generated/manifest.json` and `src/generated/<cert>.index.json` (gitignored; regenerated by npm `pre*` hooks). The app imports the manifest eagerly and loads cert indexes, 25-item question shards and lessons lazily through `import.meta.glob`, so each becomes its own chunk.

 * `edit` receives the cert object and may mutate it before it is returned.
 * @param {(cert: any) => void} [edit]

export function makeContent(edit) {

  const cert = {
    certId: 'tst-c01',
    blueprint: makeBlueprint(),
    references: [
      makeCuratedReference({ tasks: ['1.1', '1.2', '2.1'] }),
      makeCuratedReference({ url: REF_URL_2, title: 'What is cloud computing?', tasks: ['2.1'] }),

import { z } from 'zod'

/** Hosts (and their subdomains) that curated references may point to. */
export const ALLOWED_REFERENCE_HOSTS = ['aws.amazon.com', 'd1.awsstatic.com', 'repost.aws']

/** Option ids, in the order options must appear. */
export const OPTION_IDS = /** @type {const} */ (['a', 'b', 'c', 'd', 'e', 'f'])

export const MAX_SHARD_ITEMS = 25

/**
 * True for https URLs on an allowed AWS host or one of its subdomains.
 * @param {string} value
 */
export function isAllowedReferenceUrl(value) {
  let url
  try {
    url = new URL(value)
  } catch {
    return false
  }
  if (url.protocol !== 'https:') return false
  return ALLOWED_REFERENCE_HOSTS.some(
    (host) => url.hostname === host || url.hostname.endsWith(`.${host}`),
  )
}

const text = z.string().trim().min(1)
const positiveInt = z.number().int().positive()
const taskCode = z.string().regex(/^\d+\.\d+$/, 'must look like "1.1"')
const isoDate = z.string().regex(/^\d{4}-\d{2}-\d{2}$/, 'must be YYYY-MM-DD')
const awsUrl = z.string().refine(isAllowedReferenceUrl, 'must be an https URL on an allowed AWS host')

/** A reference cited by an item. */
export const ReferenceSchema = z.strictObject({ url: awsUrl, title: text })

/** One entry of content/<cert>/references.json: a link-checked page and the tasks it supports. */
export const CuratedReferenceSchema = z.strictObject({
  url: awsUrl,
  title: text,
  tasks: z.array(taskCode).min(1),
})

export const ReferencesFileSchema = z.array(CuratedReferenceSchema)

export const BlueprintSchema = z.strictObject({
  schemaVersion: z.literal(1),
  certId: z.string().regex(/^[a-z]{3}-c\d{2}$/, 'must look like "clf-c02"'),
  code: z.string().regex(/^[A-Z]{3}-C\d{2}$/, 'must look like "CLF-C02"'),
  title: text,
  displayOrder: positiveInt,
  source: z.strictObject({ url: awsUrl, retrievedOn: isoDate }),
  exam: z.strictObject({
    questions: positiveInt,
    scored: positiveInt,
    minutes: positiveInt,
    passScaled: z.number().int().min(100).max(1000),
  }),
  itemTargets: z.strictObject({ seed: positiveInt, full: positiveInt }),
  domains: z
    .array(
      z.strictObject({
        no: positiveInt,
        title: text,
        weight: z.number().gt(0).lt(1),
        subdomains: z
          .array(z.strictObject({ code: taskCode, title: text, objectives: z.array(text).min(1) }))
          .min(1),
      }),
    )
    .min(1),
})

export const OptionSchema = z.strictObject({
  id: z.enum(OPTION_IDS),
  text,
  correct: z.boolean(),
  rationale: z.string().trim().min(20),
})

export const ItemSchema = z.strictObject({
  id: z.string().regex(/^[a-z]{3}-c\d{2}-\d+\.\d+-\d{4}$/, 'must look like "clf-c02-1.1-0001"'),
  version: positiveInt,
  type: z.enum(['single', 'multiple']),
  selectCount: positiveInt,
  difficulty: z.union([z.literal(1), z.literal(2), z.literal(3)]),
  stem: z.string().trim().min(20),
  options: z.array(OptionSchema).min(4).max(6),
  explanation: z.string().trim().min(100),
  references: z.array(ReferenceSchema).min(1).max(3),
  retired: z.boolean().optional(),
})

export const ShardSchema = z.strictObject({
  schemaVersion: z.literal(1),
  subdomain: taskCode,
  items: z.array(ItemSchema).min(1).max(MAX_SHARD_ITEMS),
})
```

```js
/** Stems at or above this trigram similarity are reported as near-duplicates. */
export const NEAR_DUPLICATE_THRESHOLD = 0.8

/**
 * Character trigrams of the text, lower-cased with punctuation collapsed to spaces.
 * @param {string} text
 * @returns {Set<string>}
 */
export function trigrams(text) {
  const normalized = text
    .toLowerCase()
    .replace(/[^a-z0-9]+/g, ' ')
    .trim()
  const grams = new Set()
  for (let i = 0; i + 3 <= normalized.length; i++) grams.add(normalized.slice(i, i + 3))
  return grams
}

/**
 * Jaccard similarity of two sets: shared / union.
 * @param {Set<string>} a
 * @param {Set<string>} b
 */
export function jaccard(a, b) {
  if (a.size === 0 && b.size === 0) return 1
  let shared = 0
  for (const gram of a) if (b.has(gram)) shared++
  return shared / (a.size + b.size - shared)
}

/**
 * @param {string} a
 * @param {string} b
 */
export function stemSimilarity(a, b) {
  return jaccard(trigrams(a), trigrams(b))
}
```

/** @typedef {{ certs: CertContent[] }} Content */

/**
 * Reads every cert folder under `rootDir`. Missing folders and files come back empty so the
 * validator can explain what is missing; malformed JSON throws with the file path.
 * @param {string} rootDir
 * @returns {Promise<Content>}
 */
export async function readContent(rootDir) {
  const certIds = (await listDirNames(rootDir)).sort()
  return { certs: await Promise.all(certIds.map((id) => readCert(path.join(rootDir, id), id))) }
}

/**
 * @param {string} dir
 * @param {string} certId
 * @returns {Promise<CertContent>}
 */
async function readCert(dir, certId) {
  /** @type {ShardFile[]} */
  const shards = []
  for (const task of await listDirNames(path.join(dir, 'items'))) {
    for (const file of await listFileNames(path.join(dir, 'items', task), '.json')) {
      shards.push({ task, file, data: await readJson(path.join(dir, 'items', task, file)) })
    }
  }
  shards.sort((a, b) => compareTaskCodes(a.task, b.task) || a.file.localeCompare(b.file))

  /** @type {LessonFile[]} */
  const lessons = []
  for (const file of await listFileNames(path.join(dir, 'review'), '.md')) {
    const text = await readFile(path.join(dir, 'review', file), 'utf8')
    lessons.push({ task: file.slice(0, -'.md'.length), text })
  }
  lessons.sort((a, b) => compareTaskCodes(a.task, b.task))

  return {
    certId,
    blueprint: await readJson(path.join(dir, 'blueprint.json'), null),
    references: await readJson(path.join(dir, 'references.json'), []),
    shards,
    lessons,
  }
}

/** @param {string} dir */
async function listEntries(dir) {
  try {
    return await readdir(dir, { withFileTypes: true })
  } catch (error) {
    if (isMissing(error)) return []
    throw error
  }
}

/** @param {string} dir */
async function listDirNames(dir) {
  return (await listEntries(dir)).filter((entry) => entry.isDirectory()).map((entry) => entry.name)
}

/**
 * @param {string} dir
 * @param {string} extension
 */
async function listFileNames(dir, extension) {
  return (await listEntries(dir))
    .filter((entry) => entry.isFile() && entry.name.endsWith(extension))
    .map((entry) => entry.name)
    .sort()
}

/**
 * @param {string} file
 * @param {unknown} [fallback] returned when the file is missing; omit to require the file
 */
async function readJson(file, fallback) {
  let text
  try {
    text = await readFile(file, 'utf8')
  } catch (error) {
    if (isMissing(error) && fallback !== undefined) return fallback
    throw error
  }
  try {
    return JSON.parse(text)
  } catch (error) {
    throw new Error(`Invalid JSON in ${file}: ${/** @type {Error} */ (error).message}`)
  }
}

/** @param {any} error */
function isMissing(error) {
  return error?.code === 'ENOENT'
}

/**
 * Checks all content. Errors make content unusable (build:index refuses to run); warnings
 * flag coverage or quality gaps that `validate:content --strict` also treats as failures.
 * @param {Content} content
 * @returns {ValidationResult}
 */
export function validateContent(content) {
  /** @type {ValidationResult} */
  const result = { errors: [], warnings: [] }
  for (const cert of content.certs) validateCert(cert, result)
  return result

/**
 * @param {CertContent} cert
 * @param {ValidationResult} result
 */
function validateCert(cert, result) {
  const base = `content/${cert.certId}`
  /** @type {Report} */
  const error = (file, message) => result.errors.push({ file: `${base}/${file}`, message })
  /** @type {Report} */
  const warn = (file, message) => result.warnings.push({ file: `${base}/${file}`, message })

  const blueprint = checkBlueprint(cert, error)
  if (!blueprint) return
  const taskCodes = new Set(blueprint.domains.flatMap((d) => d.subdomains.map((s) => s.code)))
  const curated = checkReferences(cert.references, taskCodes, error)

  /** @type {Map<string, Item[]>} */
  const activeByTask = new Map([...taskCodes].map((code) => [code, []]))
  checkShards(cert, taskCodes, curated, activeByTask, error)
  const lessonTasks = checkLessons(cert, taskCodes, curated, error, warn)
  checkCoverage(blueprint, activeByTask, lessonTasks, warn)
}

/**
 * @param {import('zod').ZodError} zodError
 * @param {string} file
 * @param {Report} error
 */
function reportZod(zodError, file, error) {
  for (const issue of zodError.issues) {
    const where = issue.path.join('.')
    error(file, where ? `${where}: ${issue.message}` : issue.message)
  }
}

/**
 * @param {CertContent} cert
 * @param {Report} error
 * @returns {Blueprint | null}
 */
function checkBlueprint(cert, error) {
  if (cert.blueprint === null) {
    error('blueprint.json', 'file is missing')
    return null
  }
  const parsed = BlueprintSchema.safeParse(cert.blueprint)
  if (!parsed.success) {
    reportZod(parsed.error, 'blueprint.json', error)
    return null
  }
  const blueprint = parsed.data
  if (blueprint.certId !== cert.certId) {
    error('blueprint.json', `certId "${blueprint.certId}" must match its folder "${cert.certId}"`)
  }
  const totalWeight = blueprint.domains.reduce((sum, domain) => sum + domain.weight, 0)
  if (Math.abs(totalWeight - 1) > 0.001) {
    error('blueprint.json', `domain weights must sum to 1 (got ${totalWeight.toFixed(3)})`)
  }
  blueprint.domains.forEach((domain, d) => {
    if (domain.no !== d + 1) {
      error('blueprint.json', `domain ${d + 1} is numbered ${domain.no}; number domains 1, 2, 3…`)
    }
    domain.subdomains.forEach((sub, s) => {
      const expected = `${domain.no}.${s + 1}`
      if (sub.code !== expected) {
        error('blueprint.json', `task "${sub.code}" should be "${expected}" (tasks are numbered within their domain)`)
      }
    })
  })
  return blueprint
}

/**
 * @param {unknown} references
 * @param {Set<string>} taskCodes
 * @param {Report} error
 * @returns {Set<string>} curated URLs
 */
function checkReferences(references, taskCodes, error) {
  const parsed = ReferencesFileSchema.safeParse(references)
  if (!parsed.success) {
    reportZod(parsed.error, 'references.json', error)
    return new Set()
  }
  const urls = new Set()
  for (const ref of parsed.data) {
    if (urls.has(ref.url)) error('references.json', `duplicate URL ${ref.url}`)
    urls.add(ref.url)
    for (const task of ref.tasks) {
      if (!taskCodes.has(task)) error('references.json', `${ref.url} lists unknown task "${task}"`)
    }
  }
  return urls
}

/**
 * @param {CertContent} cert
 * @param {Set<string>} taskCodes
 * @param {Set<string>} curated
 * @param {Map<string, Item[]>} activeByTask
 * @param {Report} error
 */
function checkShards(cert, taskCodes, curated, activeByTask, error) {
  const seenIds = new Set()
  for (const [task, shards] of Map.groupBy(cert.shards, (shard) => shard.task)) {
    const folder = `items/${task}`
    if (!taskCodes.has(task)) {
      error(folder, `"${task}" is not a task in blueprint.json`)
      continue
    }
    shards.forEach((shard, s) => {
      const file = `${folder}/${shard.file}`
      const expectedName = `${String(s + 1).padStart(3, '0')}.json`
      if (shard.file !== expectedName) {
        error(file, `shard files must be numbered 001.json, 002.json… (expected ${expectedName})`)
      }
      const parsed = ShardSchema.safeParse(shard.data)
      if (!parsed.success) {
        reportZod(parsed.error, file, error)
        return
      }
      const { subdomain, items } = parsed.data
      if (subdomain !== task) error(file, `subdomain "${subdomain}" must match its folder "${task}"`)
      if (s < shards.length - 1 && items.length !== MAX_SHARD_ITEMS) {
        error(file, `holds ${items.length} items; every shard except the last must hold ${MAX_SHARD_ITEMS}`)
      }
      items.forEach((item, n) => {
        /** @param {string} message */
        const say = (message) => error(file, `items.${n} (${item.id}): ${message}`)
        if (seenIds.has(item.id)) say('duplicate id')
        seenIds.add(item.id)
        checkItem(item, `${cert.certId}-${task}-`, curated, say)
        if (!item.retired) activeByTask.get(task)?.push(item)
      })
    })
  }
}

/**
 * @param {Item} item
 * @param {string} idPrefix
 * @param {Set<string>} curated
 * @param {(message: string) => void} say
 */
function checkItem(item, idPrefix, curated, say) {
  if (!item.id.startsWith(idPrefix)) say(`id must start with "${idPrefix}"`)
  item.options.forEach((option, i) => {
    if (option.id !== OPTION_IDS[i]) {
      say(`option ${i + 1} must have id "${OPTION_IDS[i]}" (ids run a, b, c… in order)`)
    }
    if (BANNED_OPTION.test(option.text)) {
      say(`option ${option.id} uses a banned phrase ("all/none of the above")`)
    }
  })
  const correct = item.options.filter((option) => option.correct).length
  if (item.type === 'single') {
    if (item.options.length !== 4) say('single-answer items need exactly 4 options')
    if (item.selectCount !== 1) say('single-answer items need selectCount 1')
    if (correct !== 1) say(`single-answer items need exactly 1 correct option (found ${correct})`)
  } else {
    if (item.options.length < 5) say('multiple-response items need 5 or 6 options')
    if (item.selectCount < 2) say('multiple-response items need selectCount of at least 2')
    if (correct !== item.selectCount) {
      say(`selectCount is ${item.selectCount} but ${correct} options are correct`)
    }
    if (item.options.length < item.selectCount + 3) {
      say(`needs at least ${item.selectCount + 3} options for selectCount ${item.selectCount}`)
    }
  }
  if (SELECT_COUNT_IN_STEM.test(item.stem)) {
    say('remove "(Select …)" from the stem; the app shows it from selectCount')
  }
  for (const ref of item.references) {
    if (!curated.has(ref.url)) say(`reference ${ref.url} is not in references.json`)
  }
}

/**
 * @param {CertContent} cert
 * @param {Set<string>} taskCodes
 * @param {Set<string>} curated
 * @param {Report} error
 * @param {Report} warn
 * @returns {Set<string>} tasks that have a lesson
 */
function checkLessons(cert, taskCodes, curated, error, warn) {
  const lessonTasks = new Set()
  const [minWords, maxWords] = LESSON_WORD_RANGE
  for (const lesson of cert.lessons) {
    const file = `review/${lesson.task}.md`
    if (!taskCodes.has(lesson.task)) {
      error(file, `"${lesson.task}" is not a task in blueprint.json`)
      continue
    }
    lessonTasks.add(lesson.task)
    const lines = lesson.text.split(/\r?\n/)
    const firstLine = lines.find((line) => line.trim() !== '') ?? ''
    if (!firstLine.startsWith('# ')) error(file, 'must start with a level-1 heading ("# Title")')
    const fences = lines.filter((line) => line.trimStart().startsWith('```')).length
    if (fences % 2 !== 0) error(file, 'has an unclosed ``` code fence')
    for (const [, url] of lesson.text.matchAll(MARKDOWN_LINK)) {
      if (!curated.has(url)) error(file, `link ${url} is not in references.json`)
    }
    const words = lesson.text.split(/\s+/).filter(Boolean).length
    if (words < minWords || words > maxWords) {
      warn(file, `has ${words} words; aim for ${minWords}–${maxWords}`)
    }
  }
  return lessonTasks
}

/**
 * @param {Blueprint} blueprint
 * @param {Map<string, Item[]>} activeByTask
 * @param {Set<string>} lessonTasks
 * @param {Report} warn
 */
function checkCoverage(blueprint, activeByTask, lessonTasks, warn) {
  const [minShare, maxShare] = MR_SHARE_RANGE
  for (const [task, items] of activeByTask) {
    const folder = `items/${task}`
    if (!lessonTasks.has(task)) warn(`review/${task}.md`, 'lesson is missing')
    if (items.length < blueprint.itemTargets.seed) {
      warn(folder, `has ${items.length} items; the seed target is ${blueprint.itemTargets.seed}`)
    }
    if (items.length >= 10) {
      const share = items.filter((item) => item.type === 'multiple').length / items.length
      if (share < minShare || share > maxShare) {
        warn(
          folder,
          `multiple-response share is ${Math.round(share * 100)}%; aim for ${Math.round(minShare * 100)}–${Math.round(maxShare * 100)}%`,
        )
      }
    }
    const grams = items.map((item) => trigrams(item.stem))
    for (let a = 0; a < items.length; a++) {
      for (let b = a + 1; b < items.length; b++) {
        if (jaccard(grams[a], grams[b]) >= NEAR_DUPLICATE_THRESHOLD) {
          warn(folder, `${items[a].id} and ${items[b].id} have near-duplicate stems`)
        }
      }
    }
  }
}

import { domainId, subdomainId } from '../../src/content/ids.js'

/** @typedef {import('./readContent.mjs').CertContent} CertContent */
/** @typedef {import('./readContent.mjs').Content} Content */
/** @typedef {import('../../src/content/types.js').Manifest} Manifest */
/** @typedef {import('../../src/content/types.js').ManifestCert} ManifestCert */
/** @typedef {import('../../src/content/types.js').CertIndex} CertIndex */

// These builders assume the content already passed validateContent().

/**
 * Active (non-retired) item ids per task, per shard, in file order.
 * @param {CertContent} cert
 * @returns {Map<string, { shardNo: string, ids: string[] }[]>}
 */
function activeIdsByTask(cert) {
  const byTask = new Map()
  for (const shard of cert.shards) {
    const ids = shard.data.items.filter((item) => !item.retired).map((item) => item.id)
    const list = byTask.get(shard.task) ?? []
    list.push({ shardNo: shard.file.replace(/\.json$/, ''), ids })
    byTask.set(shard.task, list)
  }
  return byTask
}

/** @param {{ itemCount: number }[]} list */
const totalItems = (list) => list.reduce((sum, entry) => sum + entry.itemCount, 0)

/**
 * @param {Content} content
 * @returns {Manifest}
 */
export function buildManifest(content) {
  const certs = content.certs.map((cert) => {
    const { blueprint } = cert
    const byTask = activeIdsByTask(cert)
    const lessonTasks = new Set(cert.lessons.map((lesson) => lesson.task))
    /** @param {string} code */
    const countFor = (code) => (byTask.get(code) ?? []).reduce((n, shard) => n + shard.ids.length, 0)
    const domains = blueprint.domains.map((domain) => {
      const subdomains = domain.subdomains.map((sub) => ({
        id: subdomainId(blueprint.certId, sub.code),
        code: sub.code,
        title: sub.title,
        itemCount: countFor(sub.code),
        hasLesson: lessonTasks.has(sub.code),
      }))
      return {
        id: domainId(blueprint.certId, domain.no),
        no: domain.no,
        title: domain.title,
        weight: domain.weight,
        itemCount: totalItems(subdomains),
        subdomains,
      }
    })
    /** @type {ManifestCert} */
    const summary = {
      id: blueprint.certId,
      code: blueprint.code,
      title: blueprint.title,
      exam: blueprint.exam,
      itemCount: totalItems(domains),
      domains,
    }
    return { displayOrder: blueprint.displayOrder, summary }
  })
  certs.sort((a, b) => a.displayOrder - b.displayOrder)
  return { schemaVersion: 1, certs: certs.map((cert) => cert.summary) }
}

/**
 * @param {CertContent} cert
 * @returns {CertIndex}
 */
export function buildCertIndex(cert) {
  const byTask = activeIdsByTask(cert)
  /** @type {CertIndex} */
  const index = { objectives: {}, shards: {} }
  for (const domain of cert.blueprint.domains) {
    for (const sub of domain.subdomains) {
      index.objectives[sub.code] = sub.objectives
      const shards = (byTask.get(sub.code) ?? []).filter((shard) => shard.ids.length > 0)
      index.shards[sub.code] = Object.fromEntries(shards.map((shard) => [shard.shardNo, shard.ids]))
    }
  }
  return index
}

import { formatIssue } from './validate.mjs'

/**
 * Turns a validation result into printable lines and a process exit code.
 * @param {import('./validate.mjs').ValidationResult} result
 * @param {{ strict?: boolean }} [options] strict: warnings also fail
 * @returns {{ lines: string[], exitCode: number }}
 */
export function reportValidation({ errors, warnings }, { strict = false } = {}) {
  const summary = `${errors.length} error(s), ${warnings.length} warning(s)`
  const lines = [
    ...errors.map((issue) => `✗ ${formatIssue(issue)}`),
    ...warnings.map((issue) => `⚠ ${formatIssue(issue)}`),
    strict ? `${summary} (strict: warnings fail)` : summary,
  ]
  const failed = errors.length > 0 || (strict && warnings.length > 0)
  return { lines, exitCode: failed ? 1 : 0 }
}

/** @typedef {import('./readContent.mjs').Content} Content */

/**
 * @typedef {object} TaskStats
 * @property {string} task
 * @property {number} active non-retired items
 * @property {number} retired
 * @property {number} multiple active multiple-response items
 * @property {number[]} difficulty active items at difficulty 1, 2 and 3
 * @property {boolean} hasLesson
 */

/**
 * @typedef {object} CertStats
 * @property {string} certId
 * @property {number} seedTarget
 * @property {number} fullTarget
 * @property {TaskStats[]} tasks
 */

/**
 * Progress against the item targets. Works on unvalidated content: unreadable shards count
 * as empty.
 * @param {Content} content
 * @returns {CertStats[]}
 */
export function computeStats(content) {
  return content.certs
    .filter((cert) => Array.isArray(cert.blueprint?.domains))
    .map((cert) => {
      const lessons = new Set(cert.lessons.map((lesson) => lesson.task))
      const codes = cert.blueprint.domains.flatMap((domain) => domain.subdomains.map((sub) => sub.code))
      const tasks = codes.map((task) => {
        const items = cert.shards
          .filter((shard) => shard.task === task)
          .flatMap((shard) => (Array.isArray(shard.data?.items) ? shard.data.items : []))
        const active = items.filter((item) => !item.retired)
        return {
          task,
          active: active.length,
          retired: items.length - active.length,
          multiple: active.filter((item) => item.type === 'multiple').length,
          difficulty: [1, 2, 3].map((level) => active.filter((item) => item.difficulty === level).length),
          hasLesson: lessons.has(task),
        }
      })
      return {
        certId: cert.certId,
        seedTarget: cert.blueprint.itemTargets?.seed ?? 0,
        fullTarget: cert.blueprint.itemTargets?.full ?? 0,
        tasks,
      }
    })
}

/**
 * @param {CertStats[]} stats
 * @returns {string}
 */
export function formatStats(stats) {
  const lines = []
  for (const cert of stats) {
    const total = cert.tasks.reduce((sum, task) => sum + task.active, 0)
    lines.push(
      `${cert.certId}: ${total} active items (targets per task: seed ${cert.seedTarget}, full ${cert.fullTarget})`,
    )
    lines.push('  task   items   MR  d1/d2/d3   lesson  seed  full')
    for (const t of cert.tasks) {
      lines.push(
        [
          `  ${t.task.padEnd(6)}`,
          String(t.active).padStart(5),
          String(t.multiple).padStart(4),
          `  ${t.difficulty.join('/').padEnd(9)}`,
          t.hasLesson ? 'yes   ' : 'no    ',
          t.active >= cert.seedTarget ? '  ✓ ' : '  · ',
          t.active >= cert.fullTarget ? '   ✓' : '   ·',
        ].join(' '),
      )
    }
  }
  return lines.join('\n')
}

```js
/**
 * @typedef {object} LinkResult
 * @property {string} url
 * @property {boolean} ok
 * @property {number} status HTTP status, or 0 when the request failed outright
 * @property {string | null} finalUrl where a redirect ended, else null
 * @property {string} [error]
 */

/**
 * Checks each URL with HEAD, falling back to GET because some servers reject HEAD.
 * @param {string[]} urls
 * @param {{ fetchImpl?: (url: string, init: RequestInit) => Promise<any>, concurrency?: number, timeoutMs?: number }} [options]
 * @returns {Promise<LinkResult[]>} in the same order as `urls`
 */
export async function checkUrls(urls, { fetchImpl = fetch, concurrency = 5, timeoutMs = 15_000 } = {}) {
  /** @type {LinkResult[]} */
  const results = new Array(urls.length)
  let next = 0
  async function worker() {
    while (next < urls.length) {
      const i = next++
      results[i] = await checkUrl(urls[i], fetchImpl, timeoutMs)
    }
  }
  await Promise.all(Array.from({ length: Math.min(concurrency, urls.length) }, worker))
  return results
}

/**
 * @param {string} url
 * @param {(url: string, init: RequestInit) => Promise<any>} fetchImpl
 * @param {number} timeoutMs
 * @returns {Promise<LinkResult>}
 */
async function checkUrl(url, fetchImpl, timeoutMs) {
  /** @param {string} method */
  const request = (method) =>
    fetchImpl(url, { method, redirect: 'follow', signal: AbortSignal.timeout(timeoutMs) })
  try {
    let response = await request('HEAD')
    if (!response.ok) response = await request('GET')
    return {
      url,
      ok: response.ok,
      status: response.status,
      finalUrl: response.redirected ? response.url : null,
    }
  } catch (error) {
    return {
      url,
      ok: false,
      status: 0,
      finalUrl: null,
      error: error instanceof Error ? error.message : String(error),
    }
  }
}

```js
/**
 * Least-recently-used cache. Reading or writing a key makes it the most recent; the
 * oldest key is dropped once `capacity` is exceeded.
 * @template K, V
 */
export class LruCache {
  /** @param {number} capacity */
  constructor(capacity) {
    if (!Number.isInteger(capacity) || capacity < 1) {
      throw new RangeError('capacity must be a positive integer')
    }
    this.capacity = capacity
    /** @type {Map<K, V>} */
    this.entries = new Map()
  }

  /**
   * @param {K} key
   * @returns {V | undefined}
   */
  get(key) {
    if (!this.entries.has(key)) return undefined
    const value = /** @type {V} */ (this.entries.get(key))
    this.entries.delete(key)
    this.entries.set(key, value)
    return value
  }

  /**
   * @param {K} key
   * @param {V} value
   */
  set(key, value) {
    this.entries.delete(key)
    this.entries.set(key, value)
    if (this.entries.size > this.capacity) {
      this.entries.delete(/** @type {K} */ (this.entries.keys().next().value))
    }
  }

  /** @param {K} key */
  delete(key) {
    this.entries.delete(key)
  }

  get size() {
    return this.entries.size
  }
}

 * patterns at the bottom of this file: "../generated/<cert>.index.json" and
 * "/content/<cert>/items/<task>/<NNN>.json".

 */
export function createItemLoader({ indexLoaders, shardLoaders, cacheSize = 80 }) {
  /** @type {Map<string, Promise<CertData>>} */
  const certs = new Map()
  /** @type {LruCache<string, Promise<Shard>>} */
  const shards = new LruCache(cacheSize)

  /**
   * @param {string} certId
   * @returns {Promise<CertData>}
   */
  function certData(certId) {
    const cached = certs.get(certId)
    if (cached) return cached
    const load = indexLoaders[`../generated/${certId}.index.json`]
    if (!load) return Promise.reject(new Error(`Unknown certification: ${certId}`))
    const promise = load().then((module) => {
      const index = /** @type {CertIndex} */ (module)
      /** @type {Map<string, string>} */
      const locations = new Map()
      for (const [task, files] of Object.entries(index.shards)) {
        for (const [shardNo, ids] of Object.entries(files)) {
          for (const id of ids) locations.set(id, `/content/${certId}/items/${task}/${shardNo}.json`)
        }
      }
      return { index, locations }
    })
    certs.set(certId, promise)
    // Forget failures (e.g. offline) so the next call retries.
    promise.catch(() => certs.delete(certId))
    return promise
  }

  /**
   * @param {string} path
   * @returns {Promise<Shard>}
   */
  function shard(path) {
    const cached = shards.get(path)
    if (cached) return cached
    const load = shardLoaders[path]
    if (!load) return Promise.reject(new Error(`Missing shard: ${path}`))
    const promise = load().then((module) => /** @type {Shard} */ (module))
    shards.set(path, promise)
    promise.catch(() => shards.delete(path))
    return promise
  }

   * @param {string} certId
   * @returns {Promise<CertIndex>}

  async function getCertIndex(certId) {

    return Object.keys(files)

   * @param {string} certId

  }

### Task 10: Authoring guide

**Files:**

- [ ] **Step 1: Create `content/AUTHORING.md`**

````markdown
# Authoring guide

Rules for writing exam content. `npm run validate:content` enforces everything marked
**(checked)**. The rest are editorial rules every writer must follow.

## Files

```
content/<cert>/
├─ blueprint.json      # official domains, weights, task statements, objectives (do not edit)
├─ references.json     # curated, link-checked AWS pages: { url, title, tasks }
├─ items/<task>/001.json, 002.json, …   # shards of up to 25 questions
└─ review/<task>.md    # one lesson per task statement
```

## Workflow

1. Read the task's `objectives` in `blueprint.json` and the entries in `references.json`
   whose `tasks` include your task.
2. Write the lesson, then the questions.
3. Run `npm run format`, then `npm run validate:content`, then `npm run content:stats`.
4. Fix every error **and** every warning that names your files.
5. To see changes in the running app, run `npm run build:index` or restart `npm run dev`.

## Accuracy

- Use the service and feature names from the current exam guide (the blueprint objectives),
  for example "Amazon SageMaker AI", "Amazon Quick Sight", "AWS Business Support+".
- If you are not certain of a fact, open the relevant `references.json` page and confirm it
  before writing. This covers which service does what, plan features, and limits.
- Do not quote prices or other numbers that change often.
- Each question must have exactly one defensible answer set. Every distractor must be wrong
  for a reason the rationale can state.

## Questions

### Format (checked)

- `id`: `<cert>-<task>-NNNN`, using the next unused number. Never reuse or renumber an id.
  To withdraw a question, set `"retired": true` instead of deleting it. If you change a
  question's meaning, bump `version`.
- `type: "single"`: exactly 4 options, 1 correct, `selectCount: 1`.
- `type: "multiple"`: either 5 options with 2 correct (`selectCount: 2`), or 6 options with
  3 correct (`selectCount: 3`).
- Option ids run `a`, `b`, `c`… in order. Vary which positions hold the correct answers.
- Never use "all of the above", "none of the above", "both of the above" or "neither of the
  above".
- Never put "(Select TWO.)" or similar in the stem. The app shows it from `selectCount`.
- Minimum lengths: `stem` 20 characters, each `rationale` 20, `explanation` 100.
- `difficulty`: 1, 2 or 3.
- `references`: 1–3 entries, each copied exactly (`url` and `title`) from `references.json`.
- Shards hold 25 questions. Start `002.json` only once `001.json` is full.

### Style

- **Stems** read like the AWS exam: usually a short scenario ("A company wants to…
  Which AWS service…?"). One question per stem, no trick wording, and avoid NOT/EXCEPT.
- **Options** use parallel grammar and similar length. Distractors are plausible: real AWS
  services or concepts from the same category. No joke answers. Don't let the correct option
  be the longest or most detailed.
- **Difficulty:**
  - 1 = recall (what a service or term is).
  - 2 = apply (pick the service or concept for a stated need).
  - 3 = compare or scenario (two plausible choices; the stated requirements decide).
- **Rationale** (every option): 1–3 sentences on why this option is right, or why it is
  wrong *for this scenario*.
- **Explanation:** 60–150 words that teach the underlying concept, not just restate the answer.

### Mix for each 20-question seed batch

- 16 `single` + 4 `multiple`: three "2 of 5" and one "3 of 6". The validator warns when
  multiple response is outside 15–30% of a task with 10+ questions.
- Difficulty: six at level 1, ten at level 2, four at level 3.
- Cover every objective bullet of the task at least once. If a task has more bullets than you
  can cover well, cover the most exam-relevant ones and list the uncovered bullets in your
  report.
- No two stems may test the same fact the same way. The validator warns at ≥ 0.8 trigram
  similarity.

### Example

```json
{
  "id": "clf-c02-3.6-0001",
  "version": 1,
  "type": "single",
  "selectCount": 1,
  "difficulty": 2,
  "stem": "A company must keep audit logs for 7 years and expects to read them less than once a year. Which Amazon S3 storage class is the most cost-effective choice?",
  "options": [
    { "id": "a", "text": "S3 Standard", "correct": false, "rationale": "S3 Standard is designed for frequently accessed data and costs more to store rarely read logs." },
    { "id": "b", "text": "S3 Glacier Deep Archive", "correct": true, "rationale": "Glacier Deep Archive has the lowest storage cost and suits data read once a year or less, with retrieval in hours." },
    { "id": "c", "text": "S3 One Zone-IA", "correct": false, "rationale": "One Zone-IA stores data in a single Availability Zone and costs more than archive classes for multi-year retention." },
    { "id": "d", "text": "S3 Express One Zone", "correct": false, "rationale": "S3 Express One Zone targets single-digit millisecond access for performance-critical workloads, not archives." }
  ],
  "explanation": "Amazon S3 storage classes trade storage price against access speed and cost. Data that must be retained for years but is almost never read belongs in an archive class. S3 Glacier Deep Archive offers the lowest storage price in S3, with retrievals that take hours, which is acceptable for compliance archives.
